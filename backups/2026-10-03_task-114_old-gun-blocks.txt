==== addBeadMarker ====
-- THE BEAD A PLAYER CAN ACTUALLY SEE (task 90). Karen, 2026-09-27: "we are missing front knob or how
-- it calls where you aim and follow animal with that point".
--
-- The parts gun already has a `Bead` piece and gets nothing added. The uploaded gun is exactly ONE
-- MeshPart whose bead is PAINTED at a few pixels across, so there is nothing on it to aim with: this
-- draws one at the `Sight` attachment, which is the same point the ADS pose lines up with the view
-- axis. So the mark a player follows the animal with and the point the geometry aims are the same
-- Instance, not two numbers that agree today.
--
-- It is drawn by the VIEWMODEL and not by the weapon on purpose: it is cosmetic, client-side and
-- first-person only -- the gun other players see is unchanged, and `CanQuery = false` keeps it out of
-- every ray in the game, including this system's own occlusion cast.
local function addBeadMarker(clone: BasePart, config): boolean
	local attachment = clone:FindFirstChild(SIGHT_NAME)
	if not (attachment and attachment:IsA("Attachment")) then
		return false
	end
	for _, descendant in ipairs(clone:GetDescendants()) do
		if descendant:IsA("BasePart") and descendant.Name == BEAD_NAME then
			return false -- the gun brought its own
		end
	end
	local bead = Instance.new("Part")
	bead.Name = MARKER_NAME
	bead.Shape = Enum.PartType.Ball
	bead.Size = Vector3.new(config.BEAD_MARKER_STUDS, config.BEAD_MARKER_STUDS, config.BEAD_MARKER_STUDS)
	bead.Color = config.BEAD_MARKER_COLOR
	-- METAL, NOT NEON (task 92). Karen, after playing S2: "that red dot is not realistic, we need to
	-- improve what actually is on Beretta Parallelo" -- and what that gun has is a brass ball on the
	-- rib. A Neon bead is its own light source and cannot read as metal at any colour, so the material
	-- is a config row rather than a literal, and the fill light below is what keeps it visible.
	bead.Material = config.BEAD_MARKER_MATERIAL
	bead.Anchored = true
	bead.CanCollide = false
	bead.CanQuery = false
	bead.CanTouch = false
	bead.Massless = true
	bead.CastShadow = false
	bead.CFrame = clone.CFrame * (attachment :: Attachment).CFrame
	-- ON THE BARRELS WHEN THERE ARE BARRELS (task 96, round 2), so the piece it belongs to owns
	-- it: a bead is part of the rib, not part of the action. `hinge` places it either way, but
	-- the parenting is what says whose it is -- and with the gun in one piece, which is what
	-- the OFF branch and the parts gun are, it hangs on the Handle exactly as before.
	local onBarrels = clone:FindFirstChild(BARRELS_NAME)
	bead.Parent = if onBarrels and onBarrels:IsA("BasePart") then onBarrels else clone
	return true
end



################ src/client/Camera/Viewmodel.luau (cut 2) ################
==== sleeveLength ====
-- THE SLEEVE'S TWO NUMBERS, PER GUN (task 106, round 2), and they are functions rather than two more
-- config rows because the choice is one rule with one reason: the sleeve's far end must be behind
-- the eye in EVERY pose, and that is an ABSOLUTE distance, so a glove of a different size needs a
-- different length. The old gun's glove is 1.27 studs and sits close in; model B's is 0.67 and sits
-- further out, and in the broken-open frame a 1.6-stud sleeve ended at the lens
-- (`.screenshots/20261002T111802Z-pose-reload.png`). The overlap is how far the cylinder starts
-- INSIDE the cuff, and the new glove's own turned cuff needs nearly half a hand to be covered.
--
-- THE OLD GUN'S NUMBERS DO NOT MOVE, which is round 1's finding 1 in one line: the flag-OFF default
-- build draws exactly what it drew before this task.
function Viewmodel.sleeveLength(config): number
	if config ~= nil and config.NEW_GUN == true then
		return config.SLEEVE_LENGTH_NEW_GUN_STUDS
	end
	return config.SLEEVE_LENGTH_STUDS
end


==== handLengthAlongFingers / armDirection / sleeveSegments / sleeveOverlap ====
-- WHICH OF THE MESH'S OWN AXES THE FINGERS RUN ALONG, so the sleeve starts inside the cuff on both
-- gloves. The old pair's long axis IS its X; model B's LEFT glove's is its Z while its fingers run
-- along its Y (a palm-up C puts the forearm at an angle to the fingers, so its box is a diagonal --
-- `Assets.uniformTo`'s own reason), and `Size.X` there is not the hand's length at all.
local function handLengthAlongFingers(hand: BasePart, name: string, config): number
	if config == nil or config.NEW_GUN ~= true then
		return hand.Size.X
	end
	local axes = HandAssets.AXES[name]
	if axes == nil then
		return hand.Size.X
	end
	-- THE BOX PROJECTED ONTO THE AXIS, not the nearest side (round 1, finding 3): a box's extent along
	-- a direction is the sum of its sides times that direction's own components, and picking one side
	-- answered 0.534 for the LEFT glove where the real extent is 0.914. Half of 0.534 is 0.267, which
	-- is LESS than the 0.30 overlap -- the wrist segment's near end would have started on the finger
	-- side of the glove's centre, which is exactly what `SLEEVE_OVERLAP_NEW_GUN_STUDS`' comment
	-- forbids. Nothing on screen showed it, because that near end is inside the glove either way.
	local fingers = axes.fingers.Unit
	local size = hand.Size
	return math.abs(fingers.X) * size.X + math.abs(fingers.Y) * size.Y + math.abs(fingers.Z) * size.Z
end

-- WHICH WAY THE FOREARM LEAVES THE FRAME (task 108, round 2), in the CAMERA's own axes, or nil for
-- the old gun -- whose sleeve keeps running back along the hand's own wrist, because that pair and
-- its poses were measured together and nothing about them is being re-judged.
--
-- A SHOULDER DOES NOT MOVE WITH THE GRIP, which is the whole finding: round 1 ran the sleeve back
-- along the pose frame's -X, so the left yaw of 15 that puts the hand on the forend also swung the
-- arm out sideways and it crossed the aimed frame as a bar. Down, back and a little outboard is the
-- same vector in every pose, so the arm leaves through its own bottom corner in all of them.
function Viewmodel.armDirection(name: string, config): Vector3?
	if config == nil or config.NEW_GUN ~= true then
		return nil
	end
	if name == HandAssets.LEFT then
		return config.SLEEVE_TO_LEFT_SHOULDER
	end
	return config.SLEEVE_TO_RIGHT_SHOULDER
end

-- THE SLEEVE'S SEGMENTS, wrist -> elbow: {length, diameter} each. The old gun keeps ONE cylinder of
-- the length and diameter it always had, so its pixels do not move; model B's is cut into the taper
-- `Camera.Config.SLEEVE_TAPER_NEW_GUN_STUDS` lists, at the same TOTAL length (that number is what
-- keeps the far end behind the eye, and the far end is the one that must never be seen).
function Viewmodel.sleeveSegments(config): { { length: number, diameter: number } }
	-- NONE AT ALL ON MODEL B (task 111). Karen, 2026-10-02: "you can remove those arm extentions
	-- from both hands". The drawn sleeve exists because a hand model ends at the cuff and task 99's
	-- gloves had an open hole there; Karen's v2 pair carries its OWN turned cuff, and with the arms
	-- now running out of the bottom corners (task 108) the tube was a shape in the frame rather than
	-- a fix. The OLD gun keeps its sleeve: its glove has the hole this was built for.
	if config ~= nil and config.NEW_GUN == true then
		return {}
	end
	local total = Viewmodel.sleeveLength(config)
	local taper = if config ~= nil and config.NEW_GUN == true then config.SLEEVE_TAPER_NEW_GUN_STUDS else nil
	if taper == nil or #taper == 0 then
		return { { length = total, diameter = config.SLEEVE_DIAMETER_STUDS } }
	end
	local out = {}
	for _, diameter in ipairs(taper) do
		table.insert(out, { length = total / #taper, diameter = diameter })
	end
	return out
end

function Viewmodel.sleeveOverlap(config): number
	if config ~= nil and config.NEW_GUN == true then
		return config.SLEEVE_OVERLAP_NEW_GUN_STUDS
	end
	return config.SLEEVE_OVERLAP_STUDS
end

==== poseHands armFrame / placeSleeve ====
	-- WHICH WAY THE ARM RUNS, and it is measured off the model rather than assumed: the right hand's
	-- fingertips are at its mesh +X and the left hand's at its -X (`Camera.Config`'s HAND_LEFT_YAW
	-- comment, task 93), so the wrist -- and the sleeve -- runs the other way on each.
	-- EVERY SEGMENT, FROM ITS OWN ATTRIBUTE (task 108). `addHand` worked each one's distance out
	-- once, from the glove's own length along ITS OWN finger axis; this only has to turn that into a
	-- place. `towardFingers` is +1 once the alignment is on, because the aligned hand's +X IS the
	-- fingers on both gloves -- it is the OLD pair, drawn unaligned, that needs the two signs.
	--
	-- THE FOREARM'S OWN FRAME: at the hand, with its +X running toward that shoulder (round 2). The
	-- segments are strung along it, so a sleeve is never turned by the grip -- `armDirection` says
	-- why. For the OLD gun it is nil and the sleeve runs back along the pose frame's -X exactly as
	-- it always did, which is what keeps the flag-OFF build the one Karen has been playing.
	local function armFrame(name: string, base: CFrame): CFrame?
		local want = Viewmodel.armDirection(name, config)
		if want == nil then
			return nil
		end
		local along = camCFrame:VectorToWorldSpace(want :: Vector3).Unit
		-- Any up will do -- a cylinder is round about its own X -- as long as it is square to `along`
		-- and never parallel to it.
		local up = camCFrame.UpVector
		local across = up - along * up:Dot(along)
		if across.Magnitude < 1e-3 then
			across = camCFrame.RightVector - along * camCFrame.RightVector:Dot(along)
		end
		across = across.Unit
		return CFrame.fromMatrix(base.Position, along, across, along:Cross(across))
	end

	-- IT IS PLACED FROM THE POSE'S FRAME, NOT THE HAND'S (task 108): once the alignment is applied
	-- the hand part's own X is the MESH's X, which on the left glove is not the arm at all. For the
	-- new gun the frame is the ARM's, from `armFrame`; for the old one it is the pose's, where it
	-- was.
	local function placeSleeve(base: CFrame, hand: BasePart, towardFingers: number, arm: CFrame?)
		for _, sleeve in ipairs(hand:GetChildren()) do
			if sleeve.Name == SLEEVE_NAME and sleeve:IsA("BasePart") then
				local stored = sleeve:GetAttribute(SLEEVE_BACK_ATTRIBUTE)
				local back = if typeof(stored) == "number"
					then stored :: number
					else hand.Size.X / 2 + sleeve.Size.X / 2 - Viewmodel.sleeveOverlap(config);
				(sleeve :: BasePart).CFrame = if arm ~= nil
					then (arm :: CFrame) * CFrame.new(back, 0, 0)
					else base * CFrame.new(-towardFingers * back, 0, 0)
			end
		end
	end

==== build (the cloned gun) ====
local function build(handle: BasePart, config): Model
	local made = Instance.new("Model")
	made.Name = "DrivenHuntViewmodel"

	local clone = handle:Clone()
	clone.Name = "Handle"
	-- THE GUN'S PARTS COME WITH IT (Task 71). Since the shotgun became a side-by-side built from
	-- welded parts, the Handle is an INVISIBLE envelope and every visible piece is one of its
	-- children: destroying them, as this did, would have left the first-person gun as nothing at all
	-- while the third-person gun was the real one. One source, two views -- the clone IS how the
	-- viewmodel stays the same object as the Tool, and ServerScriptService.Weapon.Hardware is still
	-- the only writer of the original (docs/design/camera.md section 3.2).
	--
	-- EVERY PART IS ANCHORED, so the welds are not needed and are destroyed with everything else that
	-- is not a shape: an unanchored part under workspace.CurrentCamera would be simulated.
	--
	-- A SurfaceAppearance, a Texture and a Decal ARE the shape (Task 74). The uploaded shotgun is a
	-- MeshPart with a SurfaceAppearance child carrying its PBR maps, and the sweep as written would
	-- have destroyed it -- so the FIRST-PERSON gun would have lost its maps while the third-person
	-- gun kept them: "purple untextured legs" (docs/PROJECT_CONTEXT.md) visible only in ADS, only to
	-- the player holding it, and asserted by nothing. docs/design/asset-pipeline.md section 6.6
	-- predicted this exact sweep and named the fix as the camera owner's, which is this file.
	for _, descendant in ipairs(clone:GetDescendants()) do
		if descendant:IsA("BasePart") then
			descendant.Anchored = true
			descendant.CanCollide = false
			descendant.CanQuery = false
			descendant.CanTouch = false
			descendant.Massless = true
			descendant.CastShadow = false
		elseif
			not (
				descendant:IsA("Attachment")
				or descendant:IsA("SurfaceAppearance")
				or descendant:IsA("Texture")
				or descendant:IsA("Decal")
			)
		then
			descendant:Destroy() -- no scripts, no welds, no sounds: a shape and nothing else
		end
	end
	clone.Anchored = true
	clone.CanCollide = false
	clone.CanQuery = false -- see the header: an occlusion ray must not hit our own viewmodel
	clone.CanTouch = false
	clone.Massless = true
	clone.CastShadow = false
	clone.Parent = made
	made.PrimaryPart = clone
	-- FIRST-PERSON EXTRAS, AND THEY BELONG TO THE FIRST-PERSON BRANCH (review round 2, finding 1).
	-- The clone is drawn in BOTH branches -- in the OFF branch it is the ADS viewmodel Karen accepted
	-- on 2026-09-25 -- so a light or a drawn bead added here without asking lands in the camera task
	-- 89 promised not to touch: the fill light was throwing a glow on the ground behind a
	-- third-person player. Neither is any use there either: the shoulder camera's gun is lit by the
	-- same sun as everything else around it, and its sight picture is the accepted framing rather
	-- than a bead on the axis.
	if config.FIRST_PERSON then
		addBeadMarker(clone, config)
		addFillLight(clone, config)
	end

	stats.rebuilds += 1
	local parts = 0
	for _, descendant in ipairs(made:GetDescendants()) do
		if descendant:IsA("BasePart") then
			parts += 1
		end
	end
	stats.parts = parts
	-- COUNTED, NOT WARNED. A gun with no `Sight` still draws -- `Mode.aimOffset(nil, config)` falls
	-- back to the accepted framing -- but the sight picture is then a guess about somebody else's
	-- mesh, so the number says how often that happened and a spec asserts it is zero for a real gun.
	if not clone:FindFirstChild(SIGHT_NAME) then
		stats.sightMissing += 1
	end
	return made
end

				local clone = (Camera.Viewmodel.current() :: Model).PrimaryPart :: BasePart
				expect(clone:FindFirstChild("SightBead")).to.equal(nil)
			end)

			-- ...and a gun that brought its own bead gets nothing added: one bead, never two.
			posed(gunWithSight(true), function()
				Camera.Viewmodel.update(CFrame.new(0, 12, 0), stateAtBlend(1), firstPersonConfig)
				local model = Camera.Viewmodel.current()
				expect(model).to.be.ok()
				local clone = (model :: Model).PrimaryPart :: BasePart
				expect(clone:FindFirstChild("SightBead")).to.equal(nil)
				expect(clone:FindFirstChild("Bead")).to.be.ok()
			end)

			-- IT IS A METAL BEAD, NOT A RED DOT (task 92). Karen, after playing S2: "that red dot is
			-- not realistic, we need to improve what actually is on Beretta Parallelo" -- and what a
			-- 486 Parallelo carries is a small BRASS BALL on the rib. So the assertion is brass and
			-- metal: warm (red above green above blue), not saturated like the dot it replaces, and a
			-- material that is not its own light source. It is still deliberately NOT
			-- `Shotgun.CONFIG.LOOK.bead`: that off-white is the parts gun's own bead, a piece of the
			-- weapon other players see, while this is the aiming mark the camera draws for one player.
			local brass = Config.BEAD_MARKER_COLOR
			expect(brass.R > brass.G and brass.G > brass.B).to.equal(true)
			expect(brass.B * 255 > 40).to.equal(true) -- brass, not a red LED: blue is present
			expect(brass.R * 255 < 240).to.equal(true) -- and it is metal, not a glowing dot
			expect(Config.BEAD_MARKER_MATERIAL).to.equal(Enum.Material.Metal)
			expect(brass).never.to.equal(Shotgun.CONFIG.LOOK.bead)
			-- ...and the drawn bead really wears both, or the config rows above are a description of
			-- something nobody applied.

				local marker = clone:FindFirstChild("SightBead") :: BasePart
				expect(marker.Material).to.equal(Enum.Material.Metal)
				expect(marker.Color).to.equal(Config.BEAD_MARKER_COLOR)
			end)
