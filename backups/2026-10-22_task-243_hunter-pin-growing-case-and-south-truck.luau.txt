-- ---------------------------------------------------------------- 2.9 the packages

-- have built a door that hides the goods.

local STRAP_COLOR: { [string]: Color3 } = {

-- THE MODEL'S WORLD-ALIGNED BOX, computed over its own parts. `Model:GetBoundingBox` returns a box
-- in the PRIMARY PART'S basis, and several of these models' first part is a rotated one (a strap on
-- the bandolier's ring, a yawed licence leaf) -- so that box is neither axis-aligned nor the one a
-- shelf has to stand the thing on. PURE.
local function worldBox(model: Model): (Vector3, Vector3)
	local lo = Vector3.new(math.huge, math.huge, math.huge)
	local hi = Vector3.new(-math.huge, -math.huge, -math.huge)
	local any = false
	for _, part in ipairs(model:GetDescendants()) do
		if part:IsA("BasePart") then
			any = true
			local cf, size = part.CFrame, part.Size
			local reach = Vector3.new(
				math.abs(cf.RightVector.X * size.X)
					+ math.abs(cf.UpVector.X * size.Y)
					+ math.abs(cf.LookVector.X * size.Z),
				math.abs(cf.RightVector.Y * size.X)
					+ math.abs(cf.UpVector.Y * size.Y)
					+ math.abs(cf.LookVector.Y * size.Z),
				math.abs(cf.RightVector.Z * size.X)
					+ math.abs(cf.UpVector.Z * size.Y)
					+ math.abs(cf.LookVector.Z * size.Z)
			) / 2
			lo = lo:Min(cf.Position - reach)
			hi = hi:Max(cf.Position + reach)
		end
	end
	if not any then
		return Vector3.zero, Vector3.one
	end
	return (lo + hi) / 2, hi - lo
end

		if piece ~= nil then
			piece.Name = string.format("Part%d", index)
			local _, before = worldBox(piece)
			local tallest = math.max(before.X, before.Y, before.Z)
			local scale = if tallest > 0 then math.clamp(BUNDLE_ITEM_STUDS / tallest, 0.05, 1) else 1
			piece:ScaleTo(scale)
			-- MEASURED AGAIN AFTER THE SCALE, and moved by its BOX and not by its pivot: the pivot
			-- is whichever part happened to be made first, which for a leaning licence card or a
			-- stack of cartons is neither the model's centre nor its foot.
			local centre, size = worldBox(piece)
			local at = Vector3.new(-span / 2 + (index - 1) * BUNDLE_SPREAD, 0.096 + size.Y / 2, 0)
			local yaw = math.rad(if index % 2 == 0 then 14 else -11)
			piece:PivotTo(CFrame.new(at) * CFrame.Angles(0, yaw, 0) * CFrame.new(-centre) * piece:GetPivot())
			piece.Parent = parent
		end
	end
end


================================================================================
4: the fallback truck, pointing south (Lobby.Build)
================================================================================
local function truck(container: Instance, at: Vector3, cfg, name: string): Instance
	local folder = Instance.new("Folder")
	folder.Name = name
	local green = cfg.TRUCK.COLOR
	local dark = Color3.fromRGB(26, 28, 26)
	-- IT POINTS SOUTH (-Z), down the track and out of the yard: a truck that is about to leave.
	local bedZ = at.Z + 4
	part("Bed", Vector3.new(7, 3.4, 8), CFrame.new(at.X, at.Y + 4.4, bedZ), green, Enum.Material.Metal, true, folder)
	for index, dx in ipairs({ -3.4, 3.4 }) do
		part(
			string.format("BedSide%d", index),
			Vector3.new(0.4, 2.6, 8),
			CFrame.new(at.X + dx, at.Y + 7.3, bedZ),
			green,
			Enum.Material.Metal,
			false,
			folder
		)
	end
	part(
		"BedTail",
		Vector3.new(7, 2.6, 0.4),
		CFrame.new(at.X, at.Y + 7.3, bedZ + 4),
		green,
		Enum.Material.Metal,
		false,
		folder
	)
	part(
		"Cab",
		Vector3.new(7, 6.5, 6),
		CFrame.new(at.X, at.Y + 5.9, at.Z - 2.4),
		green,
		Enum.Material.Metal,
		true,
		folder
	)
	part(
		"Windscreen",
		Vector3.new(6.2, 3, 0.4),
		CFrame.new(at.X, at.Y + 7.4, at.Z - 5.3),
		Color3.fromRGB(42, 52, 56),
		Enum.Material.Glass,
		false,
		folder
	)
	part(
		"Bonnet",
		Vector3.new(7, 2.8, 4),
		CFrame.new(at.X, at.Y + 4.4, at.Z - 7.4),
		green,
		Enum.Material.Metal,
		true,
		folder
	)
	part(
		"Bumper",
		Vector3.new(7.4, 1.2, 0.8),
		CFrame.new(at.X, at.Y + 2.9, at.Z - 9.6),
		dark,
		Enum.Material.Metal,
		false,
		folder
	)
	for index = 1, cfg.TRUCK.WHEELS do
		local dx = if index % 2 == 0 then 3.6 else -3.6
		local dz = if index <= 2 then -6 else 4
		local wheel = part(
			string.format("Wheel%d", index),
			Vector3.new(2.2, 5, 5),
			CFrame.new(at.X + dx, at.Y + 2.5, at.Z + dz),
			dark,
			Enum.Material.Rubber,
			false,
			folder
		)
		wheel.Shape = Enum.PartType.Cylinder
	end
	folder.Parent = container
	return folder
end
